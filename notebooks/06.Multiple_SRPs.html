import sys
import time

from pysradb import SRAweb

db = SRAweb()

for srp in SRP_list:
    try:
        df = db.sra_metadata(srp)
        df.to_csv("{}.tsv".format(srp), sep="\t", index=False)
    except:
        sys.stderr.write("Error with {}\n".format(srp))
        time.sleep(0.5)
    time.sleep(0.5)
